
私聊格式：

Alice->server
{
    "to": "Bob",
    "message": "Good morning, Bob!"
}

server->Alice
{
    "status": "success"
}

server->Bob
{
    "from": "Alice",
    "message": "Good morning, Bob!"
}

user_list申请格式：

Alice->server
{
    "user": "Alice"
}

server->Alice
{
    "code": "SUCCESS"
    "userlist": ["Alice", "Bob", "Charlie"]
}

群聊格式：

Alice->server
{
    "message": "Hello everyone!"
}

server->Alice
{
    "status": "success"
}

server->Bob
{
    "from": "Alice",
    "message": "Hello everyone!"
}
