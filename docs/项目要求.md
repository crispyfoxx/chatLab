新同学手册：用 C++ 做一个聊天软件
你要用 C++ 从零做一个类似 QQ 的聊天软件，分三个阶段：命令行聊天 → 网页版聊天 → 文件传输。
在北大CLab上实现，需要先去开一个服务器Home • PKU CLab,镜像用Debian。

2.3 写 chat_lab/CMakeLists.txt
这里需要大家学习一下C++的编译，这里是一个CMakeLists的模板，在使用过程中学习一下如何使用CMake工具编译：
cmake_minimum_required(VERSION 3.17)
project(chat_lab CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)     # 生成 compile_commands.json，给 clangd 用

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug)    # 这个项目以调试为主：不开优化、调试信息最全
endif()

add_compile_options(-Wall -Wextra -Wpedantic)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-ferror-limit=5)     # clang：最多显示 5 条错误
elseif(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  add_compile_options(-fmax-errors=5)      # g++：同上
endif()

# 需要查内存问题的时候：cmake ... -DENABLE_ASAN=ON
option(ENABLE_ASAN "Enable AddressSanitizer and UBSan" OFF)
if(ENABLE_ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=address,undefined)
endif()

find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(nlohmann_json REQUIRED)      # 来自 apt 的 nlohmann-json3-dev

add_library(chat_core
  common/net.cpp
  common/protocol.cpp
)
target_include_directories(chat_core PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}
)
target_link_libraries(chat_core PUBLIC
  Threads::Threads
  OpenSSL::Crypto
  nlohmann_json::nlohmann_json
)

add_executable(chat_server server/main.cpp)
target_link_libraries(chat_server PRIVATE chat_core)

add_executable(chat_client client/main.cpp)
target_link_libraries(chat_client PRIVATE chat_core)

enable_testing()
add_executable(protocol_test tests/protocol_test.cpp)
target_link_libraries(protocol_test PRIVATE chat_core)
add_test(NAME protocol_test COMMAND protocol_test)

注意：chat_core 里列了 common/net.cpp 和 common/protocol.cpp，要先把这两个文件建出来（内容可以先空着），否则 CMake 会报"找不到文件"。
2.4 先编译通过

      -DCMAKE_BUILD_TYPE=Debug \
      -DCMAKE_C_COMPILER=clang -DCMAKE_CXX_COMPILER=clang++
cmake --build build -j
ctest --test-dir build --output-on-failure

看到 build/chat_server、build/chat_client、build/protocol_test 生成，就说明环境通了。
2.5 用 Docker 起客户端
分工是：服务端跑在虚拟机上，两个客户端各跑在一个容器里。这样你调服务端时可以直接用 gdb、valgrind、tcpdump，客户端则随手起一个、删一个。
为什么要用容器：容器把"系统 + 工具链 + 依赖版本"打包成镜像，换机器、换人都是同一套环境；同一个镜像想开几个实例就开几个；组里的生产环境就是把这些服务放进容器、用 K8s 编排的。
docker ps        # 先看装没装、有没有权限
# 没装的话（在你的虚拟机里）：
sudo apt update && sudo apt install -y docker.io
sudo usermod -aG docker "$USER"        # 把自己加进 docker 组，然后退出登录重新进
docker images    # 看看有没有现成的带编译工具的

镜像：~/chat_lab/docker/Dockerfile。它基于和你虚拟机一样的 Debian 12、用同一个软件源，所以容器里的工具链版本和虚拟机一致。
FROM debian:12

RUN apt-get update && apt-get install -y --no-install-recommends \
        build-essential clang cmake ninja-build gdb valgrind \
        libssl-dev nlohmann-json3-dev tcpdump iproute2 netcat-openbsd curl ca-certificates \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /work
CMD ["sleep", "infinity"]

cd ~/chat_lab
docker build -t chatlab:dev docker/
docker network create chatnet        # 只做一次

检查两边工具链是否一致（版本号应该一样；Debian 12 上通常是 g++ 12.2、clang++ 14.0.6、CMake 3.25.1、Ninja 1.11.1，以你的实际输出为准）：
echo "== 虚拟机 =="
g++ --version | head -1; clang++ --version | head -1; cmake --version | head -1; ninja --version

echo "== 容器 =="
docker run --rm chatlab:dev bash -lc \
  'g++ --version | head -1; clang++ --version | head -1; cmake --version | head -1; ninja --version'

起客户端容器：写一个脚本 ~/chat_lab/scripts/client.sh，每执行一次就起一个客户端。
以下是一个示例，从中可以学习一下 shell 脚本如何编写。
#!/usr/bin/env bash
set -e
PROJECT_DIR="$(cd "$(dirname "$0")/.." && pwd)"      # 就是 ~/chat_lab

docker run --rm -it \
  --network chatnet \
  --add-host=host.docker.internal:host-gateway \
  -v "${PROJECT_DIR}:/work" \
  -w /work \
  chatlab:dev ./build/chat_client "${1:-host.docker.internal}" "${2:-9000}"

chmod +x scripts/client.sh

这部分也自己配。

# 用编辑器打开，把 IndentWidth 改成 4、ColumnLimit 改成 100 之类，随你

Checks: 'bugprone-*,clang-analyzer-*'
WarningsAsErrors: ''
HeaderFilterRegex: '.*'

VS Code 的 clangd 配置：装 "clangd" 插件，关掉 C/C++ 插件的 IntelliSense，在建 ~/chat_lab/.vscode/settings.json：
{
  "clangd.arguments": ["--background-index", "--clang-tidy", "--completion-style=detailed"],
  "C_Cpp.intelliSenseEngine": "disabled",
  "editor.formatOnSave": true
}

clangd 靠 CMake 生成的 build/compile_commands.json 来索引代码——CMakeLists.txt 里已经打开了这个选项。改了源文件列表就重新 configure 一次，clangd 才能认识新文件。

第 3 章 阶段一：命令行聊天
3.1 你要做出的结果
做一个简单的命令行聊天软件，包含客户端和服务端，启动客户端后可看到当前哪些用户在线，并可以和在线的用户进行聊天。
服务端能在本机启动，两个客户端可以在容器上，也可以在本机部署。
群发所有人都收到；私聊只有目标收到；给不在线的人私聊会有明确的错误提示。
收到坏数据（长度是天文数字、内容不是合法结构、类型不认识）不会崩溃。
README.md 和 docs/protocol.md 写好；docs/design.md 里写明你的协议设计。
3.2 需要实现的功能
要满足的点
说明


必须能拒绝畸形输入
长度要有上限；魔数/格式不对、内容不合法、类型不认识时，要么忽略要么报错，不能崩
必须有类型区分且可扩展
以后要加新消息类型，不需要改动已有逻辑
必须有错误反馈
对方不在线、名字不存在、消息太长，都要给发送方一个明确回应，不许静默失败；"消息太长"指的是超过你自己声明的上限，1 MiB 的消息必须能通过
必须能发现掉线并清理
客户端正常退出或被 kill 后，服务端要立刻把它从在线列表里去掉（recv 返回 0 就能发现）


数据编码
文本统一 UTF-8；二进制数据不要用字符串函数处理
资源管理
用 RAII 管 fd、文件

3.3 可以自己设计的地方
帧的具体布局：分几个字段、长度字段几字节、要不要版本号、要不要魔数
负载格式：JSON、自定义文本、还是二进制
消息类型的编号与命名、每个字段叫什么
错误码怎么定义
并发模型：一连接一线程 / 线程池 / poll / epoll
在线用户表用什么数据结构
客户端是命令行交互还是简单菜单
3.4 实现时你会用到的几个事实
recv 一次可能只给你一部分数据；返回 0 表示对方关闭了连接（不是"收到空消息"）。
send 也可能只写出去一部分，所以要循环写直到写完。
往已经关闭的连接写数据会触发 SIGPIPE，系统默认直接杀掉你的进程；用 MSG_NOSIGNAL，或者忽略这个信号。
recv/send 返回 -1 时看 errno，EINTR 是"被信号打断"，重试即可。

第 4 章 阶段二：网页版聊天
4.1 你要做出的结果
两个浏览器分别注册、登录，互发消息可以看到。
在线列表正确；关掉页面后，另一边一段时间内看到它变成离线。
请求里偷偷把发送者写成别人，服务端不认。
超出你设定上限的请求体会被明确拒绝，服务端不会被拖死。
docs/design.md 里写明了实时方案、鉴权方案与接口划分。
4.2 需要实现的功能
要满足的点
说明
选择自己的实时性方案，可参考http及websocket实现

身份由服务端决定
发送者是谁，只能由服务端根据凭证判断


输入要有校验和上限
缺字段、JSON 非法、请求体过大，都要返回明确错误；上限要显式设置
实时性要真的实现
别人发消息后，页面不刷新也能看到，延迟 1 秒以内
失败必须可见
未登录、token 失效、消息太长、发送失败，页面上都要有提示



4.3 可需要考虑的一些问题
服务端没有新消息时，"挂着不回答"是怎么实现的？超时后怎么让页面马上再问一次？
浏览器怎么知道"我是谁"？凭证放在哪里、什么时候失效？
别人伪造一个凭证，你的服务端会怎么处理？
一条消息只该给一个人看，你怎么保证别人收不到？

第 5 章 阶段三：文件传输
5.1 你要做出的结果
传 50 MiB 文件，接收方下载后自己算的哈希和发送方一致。
上传有进度显示；成功和失败都有明确提示。
文件名检查。
超过大小上限的上传会被拒绝，并说明原因。
上传中途把客户端杀掉应如何处理。
传大文件时内存管理不应随文件大小线性增长。
docs/design.md 里写明了传输方式与失败处理。
5.2 需要实现的功能
要满足的点
说明
保障内存不会因文件太大而爆掉
内存管理
必须有端到端校验
接收方要能验证文件内容与发送方一致（比如哈希比对）
失败不能留半成品
先写临时文件，全部收完并校验通过后才改成正式文件
文件名不能直接使用
只取文件名本身，拒绝空名、..、路径分隔符、超长名字
必须有大小上限
超过上限直接拒绝，并说明原因
必须有进度与结果反馈
发送方知道自己传到哪了，失败知道为什么

5.3 需要考虑的地方
怎么知道"整个文件传完了"？如果少了一块，你怎么发现？
怎么证明收到的内容和发送的完全一样？
中途断了，服务端和接收方各会留下什么？会不会留下一个看起来正常的坏文件？
文件名可以怎么被利用来搞破坏？你怎么防？
进度是从哪来的？它能不能代表"服务端已经存好了"？
